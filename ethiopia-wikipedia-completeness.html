Ethiopia: Admin Boundary–Wikidata Crosswalk

Author

Carwil Bjork-James

Overview

Ethiopia has three administrative levels below the national level:

Level Name Count
Admin 1 Region 15
Admin 2 Zone 107
Admin 3 Woreda (District) 1,148

This document builds a crosswalk between the official OCHA/UN admin boundary file (maps/eth_admin_boundaries.xlsx) and Wikidata items for each level. Working top-down (Region → Zone → Woreda) lets us use parent-level QID matches to disambiguate child items that share names across regions.

The ultimate goal is a three-level crosswalk table data/eth_admin_crosswalk.rds linking each official pcode to its Wikidata QID.

Reusable helper functions

The functions below are designed to be country-agnostic and can be reused for other countries with an OCHA-style admin boundary Excel file.

Code 
# ---- load_admin_sheet -------------------------------------------------------
#' Load one admin sheet from an OCHA-style boundaries Excel file.
#' Saves the result as an RDS for downstream use and returns it invisibly.
#'
#' @param xlsx_path  Path to the Excel workbook.
#' @param sheet      Sheet name (e.g. "eth_admin1").
#' @param rds_path   Where to write the cached RDS.
#' @param key_cols   Character vector of columns to keep (NULL = keep all).
load_admin_sheet <- function(xlsx_path, sheet, rds_path, key_cols = NULL) {
  df <- read_excel(xlsx_path, sheet = sheet)
  if (!is.null(key_cols)) df <- select(df, any_of(key_cols))
  saveRDS(df, rds_path)
  message("Saved ", nrow(df), " rows → ", rds_path)
  invisible(df)
}

# ---- fetch_wikidata_cached --------------------------------------------------
#' Fetch Wikidata instances for an admin class, with RDS caching.
#'
#' Arguments mirror get_wikidata_instances(); the result is saved to
#' `cache_path` on first run and reloaded on subsequent runs.
fetch_wikidata_cached <- function(class_qid,
                                  cache_path,
                                  property       = NULL,
                                  property_names = NULL,
                                  languages      = c("en", "am", "om", "so", "ti", "aa"),
                                  limit          = 1000,
                                  batch_size     = 30,
                                  force          = FALSE) {
  if (!force && file.exists(cache_path)) {
    cached <- readRDS(cache_path)
    # Re-fetch automatically if this cache predates alias support
    if (any(str_starts(names(cached), "aliases_"))) {
      message("Loading cached data from ", cache_path)
      return(cached)
    }
    message("Cache at ", cache_path, " predates alias support; re-fetching...")
  }
  result <- get_wikidata_instances(
    class_qid      = class_qid,
    property       = property,
    property_names = property_names,
    languages      = languages,
    limit          = limit,
    batch_size     = batch_size
  )
  saveRDS(result, cache_path)
  message("Saved ", nrow(result), " rows → ", cache_path)
  result
}

# ---- normalize_place_name ---------------------------------------------------
#' Normalize a place-name vector for fuzzy matching.
#'
#' Strips parenthetical disambiguation, comma-suffixes (", Region"),
#' and a configurable set of common administrative-level words
#' ("woreda", "town", "special", etc.).
#'
#' @param x          Character vector of names.
#' @param strip_terms Extra words to remove (in addition to defaults).
normalize_place_name <- function(x, strip_terms = NULL) {
  default_strip <- c("woreda", "town", "special", "zone", "district",
                     "region", "city", "municipality")
  all_strip <- paste(
    c(default_strip, strip_terms),
    collapse = "|"
  )
  pattern <- paste0("\\b(", all_strip, ")\\b")
  x |>
    str_to_lower() |>
    str_remove_all("\\s*\\(.*?\\)") |>   # remove (TG), (Oromia), etc.
    str_remove_all(",.*$") |>            # remove ", Somali", ", SNNPR"
    str_remove_all(pattern) |>
    str_squish()
}

# ---- match_admin_names ------------------------------------------------------
#' Match official admin names to Wikidata labels in up to four rounds.
#'
#' Round 1: exact lower-case match on the primary label.
#' Round 2: normalized match (strip admin-level words, disambiguations).
#' Round 3: exact lower-case match against any Wikidata alias.
#' Round 4: normalized match against any Wikidata alias.
#'
#' Rounds 3–4 require `wd_df` to contain `aliases_*` list columns
#' (produced automatically when `entity_props` includes "aliases").
#'
#' Returns the official data frame with a `qid` column and a `match_round`
#' column ("exact", "normalized", "alias_exact", "alias_normalized", or NA).
#'
#' @param official_df   Data frame with official boundary records.
#' @param official_col  Name of the column containing official place names.
#' @param wd_df         Wikidata data frame with a `qid` column.
#' @param wd_label_col  Name of the label column in wd_df (default "label_en").
#' @param strip_terms   Extra words to strip in normalization rounds.
match_admin_names <- function(official_df,
                              official_col,
                              wd_df,
                              wd_label_col = "label_en",
                              strip_terms  = NULL) {

  off_lower <- str_to_lower(official_df[[official_col]])
  wd_lower  <- str_to_lower(wd_df[[wd_label_col]])

  off_norm  <- normalize_place_name(official_df[[official_col]], strip_terms)
  wd_norm   <- normalize_place_name(wd_df[[wd_label_col]],       strip_terms)

  # Round 1: exact lower-case
  r1 <- match(off_lower, wd_lower)
  matched_qid <- wd_df$qid[r1]
  match_round <- if_else(!is.na(r1), "exact", NA_character_)

  # Round 2: normalized (only for still-unmatched)
  r2 <- match(off_norm, wd_norm)
  fill_r2 <- is.na(matched_qid) & !is.na(r2)
  matched_qid[fill_r2] <- wd_df$qid[r2[fill_r2]]
  match_round[fill_r2] <- "normalized"

  # Rounds 3 & 4: match against Wikidata aliases
  alias_cols <- names(wd_df)[str_starts(names(wd_df), "aliases_")]
  if (length(alias_cols) > 0 && any(is.na(matched_qid))) {

    # Aliases are stored as ", "-collapsed strings; split them into one row each
    alias_long <- wd_df |>
      select(qid, all_of(alias_cols)) |>
      pivot_longer(-qid, names_to = "lang_col", values_to = "aliases_str") |>
      filter(!is.na(aliases_str)) |>
      mutate(alias = str_split(aliases_str, ",\\s*")) |>
      unnest(alias) |>
      filter(nzchar(alias)) |>
      mutate(
        alias_lower = str_to_lower(alias),
        alias_norm  = normalize_place_name(alias, strip_terms)
      )

    # Round 3: exact alias match
    unmatched_idx <- which(is.na(matched_qid))
    candidate3 <- tibble(idx = unmatched_idx, name_lower = off_lower[unmatched_idx]) |>
      left_join(alias_long |> distinct(alias_lower, qid),
                by = c("name_lower" = "alias_lower")) |>
      filter(!is.na(qid)) |>
      distinct(idx, .keep_all = TRUE)
    matched_qid[candidate3$idx] <- candidate3$qid
    match_round[candidate3$idx] <- "alias_exact"

    # Round 4: normalized alias match
    unmatched_idx2 <- which(is.na(matched_qid))
    candidate4 <- tibble(idx = unmatched_idx2, name_norm = off_norm[unmatched_idx2]) |>
      left_join(alias_long |> distinct(alias_norm, qid),
                by = c("name_norm" = "alias_norm")) |>
      filter(!is.na(qid)) |>
      distinct(idx, .keep_all = TRUE)
    matched_qid[candidate4$idx] <- candidate4$qid
    match_round[candidate4$idx] <- "alias_normalized"
  }

  official_df |>
    mutate(
      qid         = matched_qid,
      match_round = match_round
    )
}

# ---- report_match_status ----------------------------------------------------
#' Print a tidy match-status summary for one admin level.
report_match_status <- function(matched_df, level_name, n_official, n_wd) {
  n_exact      <- sum(matched_df$match_round == "exact",            na.rm = TRUE)
  n_norm       <- sum(matched_df$match_round == "normalized",       na.rm = TRUE)
  n_alias_ex   <- sum(matched_df$match_round == "alias_exact",      na.rm = TRUE)
  n_alias_norm <- sum(matched_df$match_round == "alias_normalized", na.rm = TRUE)
  n_man        <- sum(matched_df$match_round == "manual",           na.rm = TRUE)
  n_total      <- n_exact + n_norm + n_alias_ex + n_alias_norm + n_man
  n_unmatched  <- sum(is.na(matched_df$qid))

  tibble(
    Level           = level_name,
    `Official`      = n_official,
    `In WD`         = n_wd,
    `Exact`         = n_exact,
    `Normalized`    = n_norm,
    `Alias exact`   = n_alias_ex,
    `Alias norm`    = n_alias_norm,
    `Manual`        = n_man,
    `Total matched` = n_total,
    `Unmatched`     = n_unmatched
  )
}
