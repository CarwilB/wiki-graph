Bolivia: Poverty and Basic Services Trends (2001–2024)

library(tidyverse)
library(kableExtra)
library(htmltools)
source("src/gh-copy-assets.R")

This document presents a comparative overview of poverty and basic service access across Bolivia’s departments. Data can be exported in HTML or Wikitable (MediaWiki) format.

Note on temporal coverage: Water and sanitation data are available for 2001 and 2024. Poverty (NBI) data is available for 2012 and 2024.

# Load the English datasets generated from censo-2024-poverty.qmd
poverty_nbi_dept <- readRDS("data/cpv2024/poverty_nbi_dept.rds")
improved_water_dept <- readRDS("data/cpv2024/improved_water_dept.rds")
improved_sanitation_dept <- readRDS("data/cpv2024/improved_sanitation_dept.rds")
# 1. Extract Poverty (NBI) percentages (2012 and 2024)
poverty_tab <- poverty_nbi_dept |>
  select(
    department,
    poor_2012 = poor_total_2012,
    poor_2024 = poor_total_2024,
    pop_2012 = ref_pop_nbi_2012,
    pop_2024 = ref_pop_nbi_2024
  ) |>
  mutate(
    poverty_pct_2012 = (poor_2012 / pop_2012) * 100,
    poverty_pct_2024 = (poor_2024 / pop_2024) * 100,
    poverty_change = poverty_pct_2024 - poverty_pct_2012
  ) |>
  select(department, poverty_pct_2012, poverty_pct_2024, poverty_change)



# 3. Extract Water Lack (Unimproved) percentages (2001 and 2024)
water_tab <- improved_water_dept |>
  select(
    department,
    water_lack_2001 = pct_unimproved_water_2001,
    water_lack_2024 = pct_unimproved_water_2024
  ) |>
  mutate(
    water_change = water_lack_2024 - water_lack_2001
  ) |>
  select(department, water_lack_2001, water_lack_2024, water_change)

# 4. Extract Sanitation Lack (Unimproved) percentages (2001 and 2024)
sanitation_tab <- improved_sanitation_dept |>
  select(
    department,
    sanitation_lack_2001 = pct_unimproved_sanitation_2001,
    sanitation_lack_2024 = pct_unimproved_sanitation_2024
  ) |>
  mutate(
    sanitation_change = sanitation_lack_2024 - sanitation_lack_2001
  ) |>
  select(department, sanitation_lack_2001, sanitation_lack_2024, sanitation_change)

# 5. Extract national totals from raw poverty data (aggregate all departments)
country_poverty <- poverty_nbi_dept |>
  summarise(
    poor_2012 = sum(poor_total_2012),
    pop_2012 = sum(ref_pop_nbi_2012),
    poor_2024 = sum(poor_total_2024),
    pop_2024 = sum(ref_pop_nbi_2024)
  ) |>
  mutate(
    poverty_pct_2012 = (poor_2012 / pop_2012) * 100,
    poverty_pct_2024 = (poor_2024 / pop_2024) * 100,
    poverty_change = poverty_pct_2024 - poverty_pct_2012,
    department = "Bolivia"
  ) |>
  select(department, poverty_pct_2012, poverty_pct_2024, poverty_change)

country_water <- improved_water_dept |>
  summarise(
    n_unimproved_2001 = sum(n_unimproved_water_2001),
    n_total_2001 = sum(n_total_2001),
    n_unimproved_2024 = sum(n_unimproved_water_2024),
    n_total_2024 = sum(n_total_2024)
  ) |>
  mutate(
    water_lack_2001 = (n_unimproved_2001 / n_total_2001) * 100,
    water_lack_2024 = (n_unimproved_2024 / n_total_2024) * 100,
    water_change = water_lack_2024 - water_lack_2001,
    department = "Bolivia"
  ) |>
  select(department, water_lack_2001, water_lack_2024, water_change)

country_sanitation <- improved_sanitation_dept |>
  summarise(
    n_unimproved_2001 = sum(n_unimproved_sanitation_2001),
    n_total_2001 = sum(n_total_2001),
    n_unimproved_2024 = sum(n_unimproved_sanitation_2024),
    n_total_2024 = sum(n_total_2024)
  ) |>
  mutate(
    sanitation_lack_2001 = (n_unimproved_2001 / n_total_2001) * 100,
    sanitation_lack_2024 = (n_unimproved_2024 / n_total_2024) * 100,
    sanitation_change = sanitation_lack_2024 - sanitation_lack_2001,
    department = "Bolivia"
  ) |>
  select(department, sanitation_lack_2001, sanitation_lack_2024, sanitation_change)

country_row <- country_poverty |>
  left_join(country_water, by = "department") |>
  left_join(country_sanitation, by = "department")

# 6. Combine all (excluding marginality) and add country row at bottom
final_table <- poverty_tab |>
  left_join(water_tab, by = "department") |>
  left_join(sanitation_tab, by = "department") |>
  arrange(department) |>
  bind_rows(country_row) |>
  # Keep departments sorted, Bolivia always at bottom
  mutate(sort_key = if_else(department == "Bolivia", 999, 0)) |>
  arrange(sort_key, department) |>
  select(-sort_key)

# Helper for consistent formatting
format_pct <- function(x) {
  ifelse(is.na(x), "-", sprintf("%.0f%%", x))
}

format_chg <- function(x) {
  if(is.na(x)) return("-")
  prefix <- if(x > 0) "+" else ""
  paste0(prefix, sprintf("%.0f%%", x))
}

# Function to create Wikitext shade syntax
# color: "red" (poverty), "blue" (water), "green" (sanitation)
wikitext_shade <- function(value_pct, color) {
  if(is.na(value_pct)) return("-")
  # Extract numeric value and format
  numeric_val <- floor(as.numeric(gsub("%", "", value_pct)))
  sprintf("{{shade|color=%s|%s}}%%", color, numeric_val)
}

Below are poverty and basic services statistics for six representative municipalities, shown alongside their department and national context.

# Load municipality-level data (all three metrics)
poverty_nbi_muni <- readRDS("data/cpv2024/poverty_nbi_muni.rds")
improved_water_muni <- readRDS("data/cpv2024/improved_water_muni.rds")
improved_sanitation_muni <- readRDS("data/cpv2024/improved_sanitation_muni.rds")

# Extract data for 6 representative municipalities
muni_list <- list(
  list(muni = "Sucre", dept = "Chuquisaca"),
  list(muni = "La Paz", dept = "La Paz"),
  list(muni = "Cochabamba", dept = "Cochabamba"),
  list(muni = "Oruro", dept = "Oruro"),
  list(muni = "Santa Cruz", dept = "Santa Cruz"),
  list(muni = "Trinidad", dept = "Beni")
)

# Build municipality data tables with all metrics (2012, 2024, change)
muni_poverty_data <- poverty_nbi_muni |>
  filter(municipality %in% map_chr(muni_list, ~.$muni)) |>
  select(
    municipality,
    department,
    poor_2012 = poor_total_2012,
    poor_2024 = poor_total_2024,
    pop_2012 = ref_pop_nbi_2012,
    pop_2024 = ref_pop_nbi_2024
  ) |>
  mutate(
    poverty_pct_2012 = (poor_2012 / pop_2012) * 100,
    poverty_pct_2024 = (poor_2024 / pop_2024) * 100,
    poverty_change = poverty_pct_2024 - poverty_pct_2012
  ) |>
  select(municipality, department, poverty_pct_2012, poverty_pct_2024, poverty_change)

muni_water_data <- improved_water_muni |>
  filter(municipality %in% map_chr(muni_list, ~.$muni)) |>
  select(
    municipality,
    water_lack_2001 = pct_unimproved_water_2001,
    water_lack_2024 = pct_unimproved_water_2024
  ) |>
  mutate(
    water_change = water_lack_2024 - water_lack_2001
  ) |>
  select(municipality, water_lack_2001, water_lack_2024, water_change)

muni_sanitation_data <- improved_sanitation_muni |>
  filter(municipality %in% map_chr(muni_list, ~.$muni)) |>
  select(
    municipality,
    sanitation_lack_2001 = pct_unimproved_sanitation_2001,
    sanitation_lack_2024 = pct_unimproved_sanitation_2024
  ) |>
  mutate(
    sanitation_change = sanitation_lack_2024 - sanitation_lack_2001
  ) |>
  select(municipality, sanitation_lack_2001, sanitation_lack_2024, sanitation_change)

This document presents a comparative overview of poverty and basic service access across Bolivia's departments. Data can be exported in HTML or Wikitable (MediaWiki) format.

**Note on temporal coverage**: Water and sanitation data are available for 2001 and 2024. Poverty (NBI) data is available for 2012 and 2024.

```{r load-data}
# Load the English datasets generated from censo-2024-poverty.qmd
poverty_nbi_dept <- readRDS("data/cpv2024/poverty_nbi_dept.rds")
improved_water_dept <- readRDS("data/cpv2024/improved_water_dept.rds")
improved_sanitation_dept <- readRDS("data/cpv2024/improved_sanitation_dept.rds")
```

```{r process-data}
# 1. Extract Poverty (NBI) percentages (2012 and 2024)
poverty_tab <- poverty_nbi_dept |>
  select(
    department,
    poor_2012 = poor_total_2012,
    poor_2024 = poor_total_2024,
    pop_2012 = ref_pop_nbi_2012,
    pop_2024 = ref_pop_nbi_2024
  ) |>
  mutate(
    poverty_pct_2012 = (poor_2012 / pop_2012) * 100,
    poverty_pct_2024 = (poor_2024 / pop_2024) * 100,
    poverty_change = poverty_pct_2024 - poverty_pct_2012
  ) |>
  select(department, poverty_pct_2012, poverty_pct_2024, poverty_change)



# 3. Extract Water Lack (Unimproved) percentages (2001 and 2024)
water_tab <- improved_water_dept |>
  select(
    department,
    water_lack_2001 = pct_unimproved_water_2001,
    water_lack_2024 = pct_unimproved_water_2024
  ) |>
  mutate(
    water_change = water_lack_2024 - water_lack_2001
  ) |>
  select(department, water_lack_2001, water_lack_2024, water_change)

# 4. Extract Sanitation Lack (Unimproved) percentages (2001 and 2024)
sanitation_tab <- improved_sanitation_dept |>
  select(
    department,
    sanitation_lack_2001 = pct_unimproved_sanitation_2001,
    sanitation_lack_2024 = pct_unimproved_sanitation_2024
  ) |>
  mutate(
    sanitation_change = sanitation_lack_2024 - sanitation_lack_2001
  ) |>
  select(department, sanitation_lack_2001, sanitation_lack_2024, sanitation_change)

# 5. Extract national totals from raw poverty data (aggregate all departments)
country_poverty <- poverty_nbi_dept |>
  summarise(
    poor_2012 = sum(poor_total_2012),
    pop_2012 = sum(ref_pop_nbi_2012),
    poor_2024 = sum(poor_total_2024),
    pop_2024 = sum(ref_pop_nbi_2024)
  ) |>
  mutate(
    poverty_pct_2012 = (poor_2012 / pop_2012) * 100,
    poverty_pct_2024 = (poor_2024 / pop_2024) * 100,
    poverty_change = poverty_pct_2024 - poverty_pct_2012,
    department = "Bolivia"
  ) |>
  select(department, poverty_pct_2012, poverty_pct_2024, poverty_change)

country_water <- improved_water_dept |>
  summarise(
    n_unimproved_2001 = sum(n_unimproved_water_2001),
    n_total_2001 = sum(n_total_2001),
    n_unimproved_2024 = sum(n_unimproved_water_2024),
    n_total_2024 = sum(n_total_2024)
  ) |>
  mutate(
    water_lack_2001 = (n_unimproved_2001 / n_total_2001) * 100,
    water_lack_2024 = (n_unimproved_2024 / n_total_2024) * 100,
    water_change = water_lack_2024 - water_lack_2001,
    department = "Bolivia"
  ) |>
  select(department, water_lack_2001, water_lack_2024, water_change)

country_sanitation <- improved_sanitation_dept |>
  summarise(
    n_unimproved_2001 = sum(n_unimproved_sanitation_2001),
    n_total_2001 = sum(n_total_2001),
    n_unimproved_2024 = sum(n_unimproved_sanitation_2024),
    n_total_2024 = sum(n_total_2024)
  ) |>
  mutate(
    sanitation_lack_2001 = (n_unimproved_2001 / n_total_2001) * 100,
    sanitation_lack_2024 = (n_unimproved_2024 / n_total_2024) * 100,
    sanitation_change = sanitation_lack_2024 - sanitation_lack_2001,
    department = "Bolivia"
  ) |>
  select(department, sanitation_lack_2001, sanitation_lack_2024, sanitation_change)

country_row <- country_poverty |>
  left_join(country_water, by = "department") |>
  left_join(country_sanitation, by = "department")

# 6. Combine all (excluding marginality) and add country row at bottom
final_table <- poverty_tab |>
  left_join(water_tab, by = "department") |>
  left_join(sanitation_tab, by = "department") |>
  arrange(department) |>
  bind_rows(country_row) |>
  # Keep departments sorted, Bolivia always at bottom
  mutate(sort_key = if_else(department == "Bolivia", 999, 0)) |>
  arrange(sort_key, department) |>
  select(-sort_key)

# Helper for consistent formatting
format_pct <- function(x) {
  ifelse(is.na(x), "-", sprintf("%.0f%%", x))
}

format_chg <- function(x) {
  if(is.na(x)) return("-")
  prefix <- if(x > 0) "+" else ""
  paste0(prefix, sprintf("%.0f%%", x))
}

# Function to create Wikitext shade syntax
# color: "red" (poverty), "blue" (water), "green" (sanitation)
wikitext_shade <- function(value_pct, color) {
  if(is.na(value_pct)) return("-")
  # Extract numeric value and format
  numeric_val <- floor(as.numeric(gsub("%", "", value_pct)))
  sprintf("{{shade|color=%s|%s}}%%", color, numeric_val)
}
```

```{r define-wt-helpers}
# Helper functions for Wikitable formatting (defined here for use in municipality tables)
format_change_wt <- function(x) {
  if(is.na(x)) return("-")
  if(x > 0) {
    sprintf("{{Increase}} +%.0f%%", x)
  } else if(x < 0) {
    sprintf("{{Decrease}} %.0f%%", x)
  } else {
    "0.0%"
  }
}

wikitext_shade <- function(value_pct, color) {
  if(is.na(value_pct)) return("-")
  sprintf("{{shade|color=%s|%s}}", color, value_pct)
}
```
